# CreateListRequest

## Example Usage

```typescript
import { CreateListRequest } from "@versedbcom/sdk/models/operations";

let value: CreateListRequest = {
  title: "Best Batman Runs",
  description: "My favorite Batman story arcs ranked by quality.",
  entityType: "issues",
  isRanked: true,
  isPrivate: false,
  rules: {
    "version": 2,
    "match": {
      "all": [
        {
          "field": "launch_date",
          "op": "within_last",
          "value": {
            "count": 3,
            "unit": "months",
          },
        },
      ],
    },
    "sort": {
      "field": "launch_date",
      "direction": "desc",
    },
    "limit": 500,
  },
};
```
